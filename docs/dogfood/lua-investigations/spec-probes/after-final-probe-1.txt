  local file = ctx.call_tool('read_file', {path='config.json'})

  end

  -- Retry calling the agent up to 3 times if output is nil
  for attempt=1,3 do
    local agent = ctx.run_agent({
      name='retry-agent-'..attempt,
      prompt=p.prompt,
      model=p.model,
      input=file.content
    })
    if agent.failure ~= nil then
      error(agent.failure.error)
    end
    if agent.output ~= nil then
      return agent.output
    end
  end
  -- Return a default response if all retries fail
  return {delivered=false}
end
