```lua
return function(p, ctx)
  local file = ctx.call_tool('read_file', {path='config.json'})
  if file.error then
    error(file.error)
  end
  local attempts = 0
  local max_attempts = 3
  local agent_output = nil
  while attempts < max_attempts do
    local agent = ctx.run_agent({
      prompt = p.prompt,
      model = p.model,
      input = file.content
    })
    if agent.failure ~= nil then
      error(agent.failure.error)
    end
    if agent.output ~= nil then
      agent_output = agent.output
      break
    end
    attempts = attempts + 1
  end
  if agent_output == nil then
    return {delivered=false}
  else
    return agent_output
  end
end
```
